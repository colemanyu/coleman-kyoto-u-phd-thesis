\begin{algorithm*}[t] % *!
\caption{Optimize lines 11 to 14 in Algorithm~\ref{alg:psdtw-constrained} by reusing the computed sorted reaches $\tilde{\mathbbm{q}}_k$.}
\label{alg:psdtw-reuse-reaches}
\begin{algorithmic}[1]
\Statex Add the following line after line 1 in Algorithm~\ref{alg:psdtw-constrained}.
% \color{blue}
\State $r'_{\mathrm{cache}} \gets -1$
% \color{black}
\Statex Replace lines 11--14 in Algorithm~\ref{alg:psdtw-constrained} by the following lines.
% \color{blue}
\If{$r'_{\mathrm{cache}} = r'$}
    \For{$k \gets 1$ \textbf{to} $L_{\max}^C$}
        \If{$k \leq |\mathbbm{Q}|$}
            \State $e_\mathrm{prev} \gets \min(\floor{kl} + r', L^Q - 1)$
            \State $e_\mathrm{new} \gets \min(\floor{kl} + r', L^Q)$
            \If{$e_\mathrm{new} > e_\mathrm{prev}$}
                \State $\tilde{\mathbbm{q}}_k \gets \tilde{\mathbbm{q}}_k \cup q'_{e_\mathrm{new}}$ \Comment{Preserve the sorting while inserting $q'_{e_\mathrm{new}}$}
            \EndIf
        \Else
            \State $\mathbbm{q}_k \gets (q'_{\max(1, \ceil{k/l} - r')}, \ldots, q'_{\min(\floor{kl} + r', L^Q)})$ 
            % \Comment{$L^Q$ equals to $|Q'|$.}
            \State $\tilde{\mathbbm{q}}_k \gets \operatorname{sort}(\mathbbm{q}_k)$
        \EndIf
    \EndFor
\Else
    \For{$k \gets 1$ \textbf{to} $L_{\max}^C$} \Comment{Same as lines 12--14 in Algorithm~\ref{alg:psdtw-constrained}.}
        \State $\mathbbm{q}_k \gets (q'_{\max(1, \ceil{k/l} - r')}, \ldots, q'_{\min(\floor{kl} + r', L^Q)})$
        % \Comment{Construct indexed collection $\mathbbm{Q}$.}
        \State $\tilde{\mathbbm{q}}_k \gets \operatorname{sort}(\mathbbm{q}_k)$
    \EndFor
    \State $r_{\mathrm{cache}} \gets r'$
\EndIf
% \color{black}
\end{algorithmic}
\end{algorithm*}
